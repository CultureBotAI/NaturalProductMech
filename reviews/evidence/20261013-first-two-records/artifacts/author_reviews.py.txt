def load(name):
    return json.loads((ROOT / name).read_text())


def evidence(eid, kind, reference, locator, support, summary, **extra):
    return dict(evidence_id=eid, kind=kind, reference=reference, locator=locator,
                accessed_at=now(), support=support, summary=summary, **extra)


def assessment(aid, area, topic, outcome, summary, refs, target, **extra):
    return dict(assessment_id=aid, area=area, topic=topic, outcome=outcome,
                summary=summary, target_ids=[target], evidence_ids=refs, **extra)


def command_check(name, target, summary, required=True):
    receipt = load(name + '.receipt.json')
    return dict(check_id=name, name=name, status='passed' if receipt['exit_code'] == 0 else 'failed',
        required=required, summary=summary, target_ids=[target], command=shlex.join(receipt['command']),
        exit_code=receipt['exit_code'], scope_note='Actual command log: ' + receipt['log'] + '; SHA-256 ' + receipt['sha256'])
