import hashlib
import json
import shutil
import subprocess
from datetime import datetime, timezone
from pathlib import Path

import yaml

root = Path.cwd()
work = Path(__file__).parent
tree = work / 'review-tree'
base = '8b5a3ebf476c27b21d49a89d7e5684cef41a367d'
python = str(root / '.venv/bin/python')
assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=tree, text=True).strip() == base
changed = subprocess.check_output(['git', 'diff', '--name-only', base], text=True).splitlines()
untracked = subprocess.check_output(['git', 'ls-files', '--others', '--exclude-standard'], text=True).splitlines()
for relative in sorted(set(changed + untracked)):
    source = root / relative
    assert source.is_file() and not source.is_symlink()
    target = tree / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source, target)
    assert source.read_bytes() == target.read_bytes()
context = [
    'curation/mibig_producer_strains.tsv', 'curation/mibig_xref_exclusions.tsv',
    'src/naturalproductmech/curate/producer_strains.py',
    'src/naturalproductmech/curate/xref_exclusions.py',
    'src/naturalproductmech/curate/structure_corrections.py',
    'src/naturalproductmech/schema/naturalproductmech.yaml',
    'schema/record_review.yaml', 'conf/record_review.yaml',
    'docs/CURATION.md', 'docs/HARMONIZATION.md', 'docs/record-reviews.md',
    'docs/record-review-profile.md', '.claude/skills/review-yaml-record/SKILL.md',
    '.claude/skills/curate-yaml-record/references/review-checklist.md',
    'scripts/seed_from_sources.py', 'tests/test_xref_exclusions.py',
    'tests/test_yersiniabactin_review_archive.py',
    'reviews/evidence/20261013-yersiniabactin/manifest.json',
]
metadata = {'started_at': datetime.now(timezone.utc).isoformat().replace('+00:00', 'Z'), 'targets': []}
for slug in ('yersiniabactin', 'yersiniabactin-2'):
    previous = f'reviews/structured/20261013T115040Z-{slug}/review.yaml'
    old = yaml.safe_load((root / previous).read_text())
    target = old['targets'][0]
    target['owner_paths'] += [
        {'repository': old['repository'], 'path': p, 'role': 'Maintained scoped remediation input'}
        for p in context[:4]
    ]
    targetfile = work / f'{slug}-targets.json'
    targetfile.write_text(json.dumps({'targets': [target]}, indent=2))
    paths = sorted(set(context + [previous] + [p['path'] for p in target['owner_paths']]))
    command = [python, 'scripts/record_review.py', 'inspect', '--targets', str(targetfile)]
    for path in paths:
        command += ['--input', path]
    result = subprocess.run(command, cwd=tree, capture_output=True, text=True)
    (work / f'{slug}-inspect.stderr').write_text(result.stderr)
    assert result.returncode == 0, result.stderr
    inspected = json.loads(result.stdout)
    for item in inspected['source']['inputs']:
        assert hashlib.sha256((root / item['path']).read_bytes()).hexdigest() == item['sha256']
    (work / f'{slug}-inspect.json').write_text(result.stdout)
    metadata['targets'].append({'slug': slug, 'target': target, 'previous': previous})
(work / 'successor-capture.json').write_text(json.dumps(metadata, indent=2))
print('Captured exact current inputs for each owner on durable main base; primary/worktree hashes agree.')
