def now():
    return datetime.now(timezone.utc).isoformat().replace('+00:00', 'Z')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def owner(path, role):
    return dict(repository=PROJECT, path=path, role=role)


def evidence(eid, kind, reference, summary, support='context_only', accessed=None, **kwargs):
    return dict(evidence_id=eid, kind=kind, reference=reference, accessed_at=accessed or now(),
                support=support, summary=summary, **kwargs)


def assessment(tid, aid, area, topic, outcome, summary, eids, **kwargs):
    return dict(assessment_id=aid, area=area, topic=topic, outcome=outcome, summary=summary,
                target_ids=[tid], evidence_ids=eids, **kwargs)


def check(tid, cid, status, summary, required=True, **kwargs):
    return dict(check_id=cid, name=cid, status=status, required=required, summary=summary,
                target_ids=[tid], **kwargs)


def fresh_check(tid, name):
    r = json.loads((ROOT / (name + '-fresh.receipt.json')).read_text())
    summary = r['log'].strip()
    if name == 'labels':
        summary = r['log'][-2400:].strip()
    if name == 'queue':
        with (ROOT / 'review-queue.tsv').open() as stream:
            counts = Counter(row['queue'] for row in csv.DictReader(stream, delimiter='\t'))
        summary = 'Native curation queue regenerated without changing corpus status: ' + json.dumps(dict(counts)) + '. This queue is not scientific review coverage.'
    return check(tid, name, 'passed' if r['exit_code'] == 0 else 'failed',
                 summary,
                 command=shlex.join(r['command']), exit_code=r['exit_code'],
                 scope_note='Actual fresh run from ' + r['started_at'] + ' to ' + r['finished_at'] +
                 '; output SHA-256 ' + r['log_sha256'] + '. Results retained here, not dependent on the temporary log. ' +
                 ('Reporting mode is non-blocking; exit zero does not mean all labels match. Unsupported adapters remain skipped.' if name == 'labels' else 'Deterministic check, not scientific sign-off.'))
