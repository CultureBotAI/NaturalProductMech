import hashlib
import json
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

ROOT=Path(__file__).parent
for name,args in [
    ('provenance-fresh',['scripts/check_provenance.py']),
    ('contract-before',['-m','pytest','-q','tests/test_record_review_contract.py','tests/test_review_evidence_archive.py']),
    ('queue-fresh',['scripts/curation_worklist.py','--review-records','--limit','0','--tsv',str(ROOT/'queue-fresh.tsv')]),
]:
    start=datetime.now(timezone.utc).isoformat()
    with (ROOT/(name+'.log')).open('x') as stream:
        r=subprocess.run([sys.executable]+args,stdout=stream,stderr=subprocess.STDOUT)
    receipt=dict(command=[sys.executable]+args,started_at=start,finished_at=datetime.now(timezone.utc).isoformat(),
        exit_code=r.returncode,log=str(ROOT/(name+'.log')),
        log_sha256=hashlib.sha256((ROOT/(name+'.log')).read_bytes()).hexdigest())
    with (ROOT/(name+'.receipt.json')).open('x') as stream:
        json.dump(receipt,stream,indent=2)
    print(json.dumps(receipt),flush=True)
    assert r.returncode==0
