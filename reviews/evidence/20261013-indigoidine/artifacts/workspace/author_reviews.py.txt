def owner(path,role):

def evidence(eid,kind,reference,summary,support='partial',locator=None,path=None,accessed=None,search=None):
    d=dict(evidence_id=eid,kind=kind,reference=reference,accessed_at=accessed or now(),support=support,summary=summary)
    if locator:d['locator']=locator
    if path:d['snapshot_sha256']=sha(Path(path))
    if search:d['search_scope']=search
    return d
