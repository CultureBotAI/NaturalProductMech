import hashlib
import json
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

import yaml
from naturalproductmech.graph_components import read_natural_product

ROOT = Path(__file__).parent
REPO = Path.cwd()
OLD = Path('/private/tmp/npmech-review-indigoidine-20261013-IAW8jG')
CHECKPOINT = REPO / 'reviews/evidence/20261013-2mib-heme/artifacts/workspace/review-campaign-checkpoint.json'
BASE = '127b18b43ac6c9ac76fd4ad878979ed8ddb0de8f'


def now():
    return datetime.now(timezone.utc).isoformat().replace('+00:00', 'Z')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(name, value):
    with (ROOT / name).open('x') as stream:
        json.dump(value, stream, indent=2)


def run(name, args):
    start = now()
    path = ROOT / (name + '.log')
    with path.open('x') as stream:
        result = subprocess.run([sys.executable] + args, stdout=stream, stderr=subprocess.STDOUT)
    receipt = dict(command=[sys.executable] + args, started_at=start, finished_at=now(),
                   exit_code=result.returncode, log=str(path), sha256=sha(path))
    save(name + '.receipt.json', receipt)
    print(json.dumps(receipt), flush=True)
    assert result.returncode == 0


baseline = json.loads(CHECKPOINT.read_text())
population = {r['path']: r for r in baseline['records']}
assert set(population) == {str(p.relative_to(REPO)) for p in (REPO / 'data/natural_products').rglob('*.yaml')}
components = {c['path']: c['sha256'] for r in population.values() for c in r['graph_components']}
assert set(components) == {str(p.relative_to(REPO)) for p in (REPO / 'data/causal_graphs').rglob('*.yaml')}
for row in population.values():
    assert sha(REPO / row['path']) == row['sha256']
for path, expected in components.items():
    assert sha(REPO / path) == expected
for row in baseline['reviews']:
    assert sha(REPO / row['yaml']) == row['yaml_sha256']
    assert sha(REPO / row['markdown']) == row['markdown_sha256']
assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip() == BASE
