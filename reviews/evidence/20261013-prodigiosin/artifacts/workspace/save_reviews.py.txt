class ReviewDumper(yaml.SafeDumper):
    def ignore_aliases(self, data):
        return True


def owned(path, role):
    return dict(repository=repo, path=path, role=role)


def ev(eid, kind, reference, summary, support='partial', locator=None, snapshot=None):
    item = dict(evidence_id=eid, kind=kind, reference=reference, accessed_at=now(), support=support, summary=summary)
    if locator:
        item['locator'] = locator
    if snapshot:
        item['snapshot_sha256'] = sha(work / snapshot)
    return item


def receipt_check(label, target, summary, required=True, status=None):
    receipt = json.loads((work / f'{label}.receipt.json').read_text())
    return dict(check_id=label, name=label, status=('passed' if receipt['exit_code'] == 0 else 'failed'),
                required=required, summary=summary, command=shlex.join(receipt['command']),
                exit_code=receipt['exit_code'], target_ids=[target], evidence_ids=['receipts'])

def base_review(slug):
    inspection = json.loads((work / f'{slug}-inspect.json').read_text())
    target = inspection['targets'][0]
    tid = target['target_id']
    for item in inspection['source']['inputs']:
        assert sha(root / item['path']) == item['sha256']
    source = copy.deepcopy(inspection['source'])
    source['state'] = 'git_commit'
    for item in source['inputs']:
        committed = subprocess.check_output(['git', 'show', f"{source['git_revision']}:{item['path']}"])
        assert hashlib.sha256(committed).hexdigest() == item['sha256']
    checks = [receipt_check(f'{slug}-inspect', tid, 'Canonical target and maintained-input capture before assessment.'),
              receipt_check(f'strict-{slug}', tid, 'Initial just invocation could not open the default uv cache. Superseded by successful direct native validator.', False, 'unavailable'),
              receipt_check(f'strict-{slug}-retry', tid, 'Temporary-cache retry could not resolve hatchling without network. Superseded by successful direct native validator.', False, 'unavailable'),
              receipt_check(f'strict-{slug}-direct', tid, 'Native closed-schema validator: one owner, zero errors; complete-mechanism reader resolved zero graphs.'),
              receipt_check('qc', tid, 'Initial just qc could not open the default uv cache; superseded by the permitted retry.', False, 'unavailable'),
              receipt_check('qc-retry', tid, 'Full native quality gate passed: 623 tests passed, 3 skipped; all 3115 owners validate and reproduce. Deterministic validation does not certify the scientific claims.'),
              receipt_check('corpus', tid, 'Documented complete corpus consistency check; not a sampled substitute.'),
              receipt_check('queue', tid, 'Fresh exhaustive native review queue, retained in private workspace; no native status edits.'),
              receipt_check('labels', tid, 'Exit zero with 499 corpus warning mismatches and 4649 skipped-no-adapter pairs; not a universal identifier pass.'),
              receipt_check('pinned-chemistry', tid, 'Exact pinned RDKit version has no macOS x86_64 wheel; chemical recalculation unavailable.', True, 'unavailable'),
              dict(check_id='primary-completeness', name='Every nearest primary source and exact chemical form', status='unavailable', required=True,
                   summary='Access stops and uninspected full texts prevent complete occurrence, strain, and exact chemical-form adjudication.', target_ids=[tid]),
              dict(check_id='imodulondb', name='Structured expression-module cross-check', status='not_applicable', required=False,
                   summary='No gene, regulator, protein-example, pathway-step or transcriptomic claim is encoded in this owner. Official MIBiG structured annotations were inspected for the BGC assertions; expression-module membership cannot establish chemical occurrence.', target_ids=[tid])]
    evidence = [
        ev('owner', 'record_content', target['path'], 'Entire owner and audit trail read. No inline or referenced graphs; no quantitative assays, molecular targets, clinical claims or datasets. Native SEEDED status retained.', 'partial', 'Complete YAML, including curation_history'),
        ev('inventory', 'record_content', str(work / 'inventory-audit.json'), 'Structured scan of complete six TSV inventories retains exact row numbers and keys; four live versioned annotations match stored compound SMILES, taxon, genome accession and active status.', snapshot='inventory-audit.json'),
        ev('receipts', 'validation', str(work), 'Actual inspect, strict, corpus, queue, full-QC, label and chemistry execution receipts preserve commands, UTC timing, exit codes and log hashes. Initial environment failures are retained, not counted as passes.'),
        ev('capture', 'validation', str(work / 'capture.json'), 'Ignored-inclusive Path.rglob inventory and SHA256 checks confirm all 3115 owners, 243 components and nine prior review observations match the post-merge checkpoint.', snapshot='capture.json'),
        ev('rubric', 'record_content', '.claude/skills/curate-yaml-record/references/review-checklist.md', 'Exact compound/taxon and claim-level evidence rules, strain scope, generated ownership, and native REVIEWED gates.', 'supports'),
    ]
    evidence[0]['snapshot_sha256'] = sha(root / target['path'])
    evidence[1]['search_scope'] = 'Complete named TSVs parsed with csv.DictReader; ignored files included. Both owner files read completely. Relevant curation paths searched with rg --hidden --no-ignore; no record-only correction is assumed.'
    return dict(schema_version='1.0.0', kind='record', repository=repo,
        started_at=capture['started_at'], reviewer=dict(identity='Codex', kind='agent', independence='self_review',
        independence_basis='Same assistant/thread as prior repository work; direct source-checking pass, not an independent human or agent review.'),

                   selection=f'Next pending exact path in the exhaustive campaign checkpoint: {target["path"]}.',

def assessment(review, aid, area, topic, outcome, summary, evidence, details=None):
    item = dict(assessment_id=aid, area=area, topic=topic, outcome=outcome, summary=summary,
                target_ids=[review['targets'][0]['target_id']], evidence_ids=evidence)
    if details:
        item['details'] = details
    review['assessments'].append(item)


def finding(review, fid, key, severity, title, description, fields, evidence, owners):
    review['findings'].append(dict(finding_id=fid, issue_key=key, category='evidence' if fid == 'F1' else 'scope',
        severity=severity, status='open', certainty='confirmed', title=title, description=description,
        target_ids=[review['targets'][0]['target_id']], field_paths=fields, evidence_ids=evidence, owner_paths=owners,
        rule_id='Native checklist: exact compound and taxon, narrow evidence, producer strain where supplied',
        native_severity=severity, normalization_reason=('Wrong claim-to-source attribution changes biological interpretation while deterministic gates pass.' if severity == 'major' else 'Bounded loss of source strain provenance; producer evidence grade is not falsely upgraded.')))


def action(review, aid, finding_ids, description, owners, checks):
    review['actions'].append(dict(action_id=aid, finding_ids=finding_ids,
        target_ids=[review['targets'][0]['target_id']], description=description, owner_paths=owners,
        generator='scripts/seed_from_sources.py', acceptance_checks=checks))

saved = []
for slug, review in [('prodigiosin-2', first), ('prodigiosin', second)]:
    if slug == 'prodigiosin-2':
        existing = root / 'reviews/structured/20261013T085546Z-prodigiosin-2/review.yaml'
        prior = yaml.safe_load(existing.read_text())
        assert prior['targets'] == review['targets'] and prior['source'] == review['source']
        saved.append(dict(review_id=prior['review_id'], path=str(existing.relative_to(root)),
                          owner=prior['targets'][0]['path'], verdict=prior['verdict'], completion=prior['completion']))
        continue
    review['finished_at'] = now()
    stamp = datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%SZ')
    review['review_id'] = f'{stamp}-{slug}'
    path = work / f'{review["review_id"]}.yaml'
    with path.open('x') as f:
        yaml.dump(review, f, Dumper=ReviewDumper, sort_keys=False, allow_unicode=False, width=96)
    for command in ([sys.executable, 'scripts/record_review.py', 'validate', str(path)],
                    [sys.executable, 'scripts/record_review.py', 'save', '--content', str(path)]):
        started = now()
        result = subprocess.run(command, text=True, capture_output=True)
        label = review['review_id'] + '-' + command[2]
        log = work / (label + '.log')
        log.write_text(result.stdout + result.stderr)
        (work / (label + '.receipt.json')).write_text(json.dumps(dict(
            command=command, started_at=started, finished_at=now(), exit_code=result.returncode,
            log=str(log), log_sha256=sha(log)), indent=2))
        print(result.stdout, result.stderr, flush=True)
        assert result.returncode == 0
    saved.append(dict(review_id=review['review_id'], path=f'reviews/structured/{review["review_id"]}/review.yaml',
                      owner=review['targets'][0]['path'], verdict=review['verdict'], completion=review['completion']))
with (work / 'saved-reviews.json').open('x') as f:
    json.dump(saved, f, indent=2)
print(json.dumps(saved, indent=2))
