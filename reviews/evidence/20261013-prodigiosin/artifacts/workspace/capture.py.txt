import hashlib
import json
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

import yaml
from naturalproductmech.graph_components import read_natural_product

root = Path.cwd()
work = Path(__file__).parent
previous_path = Path('/private/tmp/npmech-mpa-review-current-NPq2SH/post-merge-campaign-checkpoint.json')
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
now = lambda: datetime.now(timezone.utc).isoformat().replace('+00:00', 'Z')
started = now()
previous = json.loads(previous_path.read_text())
assert sha(previous_path) == '54c8e52f7f631b102a9a63b86f929c65724d8f477a9a0f288d698374b3b7fa68'
base = subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip()
assert base == previous['base']
assert {r['path'] for r in previous['records']} == {str(p.relative_to(root)) for p in (root / 'data/natural_products').rglob('*.yaml')}
components = {c['path']: c['sha256'] for r in previous['records'] for c in r['graph_components']}
assert set(components) == {str(p.relative_to(root)) for p in (root / 'data/causal_graphs').rglob('*.yaml')}
for r in previous['records']:
    assert sha(root / r['path']) == r['sha256']
for p, digest in components.items():
    assert sha(root / p) == digest
for r in previous['all_review_observations']:
    assert sha(root / r['yaml']) == r['yaml_sha256']
    assert sha(root / r['markdown']) == r['markdown_sha256']

owners = {
    'data/raw/mibig_compounds.tsv': 'Imported exact chemical, producer and BGC assertions',
    'data/raw/lotus_occurrences.tsv': 'Imported organism-reference triples',
    'data/raw/chebi_structures.tsv': 'Imported ChEBI identity and definition',
    'data/raw/chebi_origins.tsv': 'Imported ChEBI occurrence assertions',
    'data/raw/npclassifier.tsv': 'Pinned computed classification',
    'data/raw/antibioticmech_inchikeys.tsv': 'Pinned same-structure sibling inventory',
    'scripts/seed_from_sources.py': 'Harmonization and generated record emission',
    'scripts/extract_lotus.py': 'Occurrence source extraction',
    'scripts/extract_mibig.py': 'Cluster source extraction',
}
inputs = [
    'docs/CURATION.md', 'docs/HARMONIZATION.md', 'docs/record-review-profile.md',
    'docs/record-reviews.md', 'CLAUDE.md', 'justfile', 'conf/record_review.yaml',
    '.claude/skills/review-yaml-record/SKILL.md',
    '.claude/skills/curate-yaml-record/references/review-checklist.md',
    'src/naturalproductmech/schema/naturalproductmech.yaml',
    'src/naturalproductmech/schema/mech_shared.yaml',
    'src/naturalproductmech/schema/history.yaml', 'schema/record_review.yaml',
    'conf/producer_evidence.tsv', 'conf/id_label_targets.yaml', 'data/raw/corpus_taxa.obo',
    'data/raw/MANIFEST.yaml', 'data/natural_products/PATHS.tsv',
    'curation/decisions.tsv', 'curation/mibig_locus_evidence_overrides.tsv',
    'curation/mibig_structure_corrections.tsv', 'pyproject.toml', 'uv.lock',
    'data/natural_products/alkaloids/prodigiosin.yaml',
    'data/natural_products/alkaloids/prodigiosin-2.yaml',
]
captures = []
for slug in ('prodigiosin-2', 'prodigiosin'):
    path = f'data/natural_products/alkaloids/{slug}.yaml'
    doc = yaml.safe_load((root / path).read_text())
    target = dict(target_id=doc['identifier'], path=path, label=doc['label'],
                  record_class='NaturalProductRecord', kind='generated',
                  owner_paths=[dict(repository='CultureBotAI/NaturalProductMech', path=p, role=r) for p, r in owners.items()])
    target_path = work / f'{slug}-targets.json'
    target_path.write_text(json.dumps(dict(targets=[target]), indent=2))
    cmd = [sys.executable, 'scripts/record_review.py', 'inspect', '--targets', str(target_path)]
    for p in inputs:
        cmd.extend(['--input', p])
    before = now()
    result = subprocess.run(cmd, text=True, capture_output=True)
    log = work / f'{slug}-inspect.json'
    log.write_text(result.stdout)
    (work / f'{slug}-inspect.stderr').write_text(result.stderr)
    assert result.returncode == 0, result.stderr
    receipt = dict(command=cmd, started_at=before, finished_at=now(), exit_code=result.returncode, log_sha256=sha(log))
    (work / f'{slug}-inspect.receipt.json').write_text(json.dumps(receipt, indent=2))
    resolved = read_natural_product(root / path)
    (work / f'{slug}-resolved.json').write_text(json.dumps(resolved, indent=2))
    captures.append(dict(slug=slug, target=target, owner_sha256=sha(root / path),
                         graphs=len(resolved.get('causal_graphs', [])), components=doc.get('causal_graph_refs', [])))
capture = dict(started_at=started, finished_at=now(), base=base, population=3115, graph_components=243,
               prior_checkpoint=str(previous_path), prior_checkpoint_sha256=sha(previous_path),
               ignored_inclusive_population_hashes_all_unchanged=True, captures=captures,
               previous_turn_classification='progress: merged PR879 and addressed five issues with exact-tree and branch-cleanup verification')
(work / 'capture.json').write_text(json.dumps(capture, indent=2))
print(json.dumps(capture, indent=2))
